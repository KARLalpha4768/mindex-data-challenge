# Everything needed to run and verify this submission from a clean clone:
#
#     pip install -r requirements.txt
#     python scripts/verify_submission.py
#
# WHY the list is this short: the pipeline is pandas plus the standard library
# (sqlite3, json, csv, argparse, dataclasses, enum, pathlib). Fewer dependencies means
# the reviewer can run it on a clean machine in one command, and it means no
# third-party library gets to make a silent type-coercion decision on our behalf --
# which is the entire theme of this submission.
#
# This file is the superset of solution/requirements.txt: identical runtime pins, plus
# pytest, because the root entry point (scripts/verify_submission.py) runs the test
# suite as part of verification rather than treating it as optional.

pandas>=2.0,<3.0        # DataFrame engine. Pinned <3.0 because pandas 3 changes
                        # string/NA semantics, and this pipeline's defect detection
                        # depends on exact null and dtype behaviour.
numpy>=1.24             # Transitively required by pandas; imported directly by
                        # solution/src/io_utils.py to serialise numpy scalars to JSON.
pytest>=7.4             # The 87-test suite in solution/tests/, and the live test-count
                        # probe in solution/scripts/check_readme_numbers.py.

# Optional, not required to run or verify anything:
