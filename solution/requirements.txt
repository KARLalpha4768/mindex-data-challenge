# means the reviewer can run this on a clean machine in one command, and it
# means no third-party library gets to make a silent type-coercion decision on
# our behalf -- which is the entire theme of this submission.

pandas>=2.0,<3.0        # DataFrame engine. Pinned <3.0 because pandas 3 changes
                        # string/NA semantics, and this pipeline's defect
                        # detection depends on exact null and dtype behaviour.
numpy>=1.24             # Transitively required by pandas; imported directly by
                        # src/io_utils.py to serialise numpy scalars to JSON.
